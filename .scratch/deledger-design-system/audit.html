const page=figma.root.children.find(p=>p.name==='03 Screens & Prototype');await figma.setCurrentPageAsync(page);
const obsolete=await figma.getNodeByIdAsync('15:21985');if(obsolete){obsolete.remove()}
const roots=page.children.filter(n=>n.type==='FRAME'&&n.visible), rootById=Object.fromEntries(roots.map(n=>[n.id,n])), adjacency={};let links=0;const broken=[],outside=[],missingFonts=[],images=[],mobileNav=[];
function top(n){while(n.parent&&n.parent.type!=='PAGE')n=n.parent;return n}
for(const root of roots){adjacency[root.id]=[];for(const n of [root,...root.findAll(n=>'reactions'in n&&n.reactions.length)])for(const r of n.reactions||[])for(const a of r.actions||[]){links++;if(a.destinationId){const dest=figma.getNodeById(a.destinationId);if(!dest)broken.push({source:n.id,to:a.destinationId});else{const topDest=top(dest);adjacency[root.id].push(topDest.id);if(topDest.parent.id!==page.id)outside.push(n.id)}}}const nav=root.children.find(n=>n.name==='Fixed bottom navigation');if(nav&&!root.name.includes(' / Modal '))mobileNav.push({id:nav.id,root:root.name,width:nav.width,y:nav.y,height:nav.height,fixed:root.numberOfFixedChildren})}
for(const t of page.findAllWithCriteria({types:['TEXT']}))if(t.hasMissingFont)missingFonts.push(t.id);
for(const n of page.findAll(n=>'fills'in n&&Array.isArray(n.fills)&&n.fills.some(p=>p.type==='IMAGE')))images.push(n.id);
const flows=page.flowStartingPoints,reachable=new Set(flows.map(f=>f.nodeId)),queue=[...reachable];while(queue.length){const id=queue.shift();for(const dest of adjacency[id]||[])if(!reachable.has(dest)){reachable.add(dest);queue.push(dest)}}
const orphanRoots=roots.filter(n=>!reachable.has(n.id)).map(n=>({id:n.id,name:n.name}));
const sourceScreens=roots.filter(n=>!n.name.includes(' / Modal ')&&!n.name.includes(' / Calculation '));
const collections=await figma.variables.getLocalVariableCollectionsAsync(),variables=await figma.variables.getLocalVariablesAsync();
const comp=figma.root.children.find(n=>n.name==='02 Components');
print(JSON.stringify({status:broken.length||outside.length||missingFonts.length||images.length||orphanRoots.length||mobileNav.some(n=>Math.abs(n.width-390)>1||Math.abs(n.y+n.height-844)>1||n.fixed<1)?'FAIL':'PASS',fileKey:figma.fileKey,pageId:page.id,stats:{frames:roots.length,screens:sourceScreens.length,modalScenes:roots.filter(n=>n.name.includes(' / Modal ')).length,calculationScenes:roots.filter(n=>n.name.includes(' / Calculation ')).length,connections:links,flows:flows.length,reachableFrames:reachable.size,componentSets:comp.children.filter(n=>n.type==='COMPONENT_SET').length,componentVariants:comp.findAllWithCriteria({types:['COMPONENT']}).length,textStyles:(await figma.getLocalTextStylesAsync()).length,variables:variables.length,collections:collections.map(c=>({name:c.name,modes:c.modes.length,variables:c.variableIds.length}))},broken,outside,missingFonts,images,mobileNavInvalid:mobileNav.filter(n=>Math.abs(n.width-390)>1||Math.abs(n.y+n.height-844)>1||n.fixed<1),orphanRoots,flows,sourceScreens:sourceScreens.map(n=>({id:n.id,name:n.name})),graph:Object.fromEntries(Object.entries(adjacency).map(([k,v])=>[k,[...new Set(v)]]))}));
